Example: Weekly Demand + Competitor Complaint Digest (Python)

import os
from apify_client import ApifyClient

client = ApifyClient(os.environ["APIFY_TOKEN"])

# 1. Track keyword momentum + daily breaking topics on Google Trends
trends_run = client.actor("leopoldsaint/google-trends-scraper-fast").call(run_input={
    "mode": "both",
    "keywords": ["ai agents", "shopify", "matcha latte"],
    "geo": "US",
    "timeRange": "today 12-m",
    "includeGeoBreakdown": True,
    "includeRelatedQueries": True,
    "maxTrendingItems": 10,
})
trends = client.dataset(trends_run["defaultDatasetId"]).list_items().items

# 2. Decode Google News headlines into real publisher URLs (no news.google.com redirects!)
news_run = client.actor("leopoldsaint/google-news-scraper-fast").call(run_input={
    "queries": ["AI agents funding"],
    "country": "US",
    "timeRange": "24h",
    "decodeUrls": True,
    "maxTotalArticles": 20,
})
news = client.dataset(news_run["defaultDatasetId"]).list_items().items

# 3. Mine 1-2 star reviews across BOTH Apple App Store & Google Play in one schema
reviews_run = client.actor("leopoldsaint/app-store-google-play-scraper").call(run_input={
    "platform": "both",
    "mode": "apps_and_reviews",
    "iosAppIds": ["570060128"],
    "androidAppIds": ["com.duolingo"],
    "minReviewRating": 1,
    "maxReviewRating": 2,
    "maxReviewsPerApp": 25,
    "country": "US",
})
reviews = client.dataset(reviews_run["defaultDatasetId"]).list_items().items

print(f"Trends: {len(trends)} | Decoded News: {len(news)} | 1-2 Star App Reviews: {len(reviews)}")

  • Star-filtered reviews prevent wasted spend: Setting minReviewRating: 1 and maxReviewRating: 2 on the App Store & Google Play Scraper only pushes and charges for negative reviews — letting product managers and indie builders extract exact churn triggers across iOS and Android simultaneously.
